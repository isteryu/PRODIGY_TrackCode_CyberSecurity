import os

def caesar_cipher(text, shift):
    """Core logic to shift characters."""
    result = ""
    for char in text:
        if char.isupper():
            result += chr((ord(char) + shift - 65) % 26 + 65)
        elif char.islower():
            result += chr((ord(char) + shift - 97) % 26 + 97)
        else:
            result += char
    return result

def process_file(input_filename, shift, mode):
    """Reads from a file, processes the text, and writes to a new file."""
    try:
        if not os.path.exists(input_filename):
            print(f"Error: The file '{input_filename}' was not found.")
            return

        with open(input_filename, 'r') as file:
            content = file.read()

        # Adjust shift for decryption
        actual_shift = -shift if mode == 'decrypt' else shift
        processed_text = caesar_cipher(content, actual_shift)

        output_filename = f"{mode}ed_{input_filename}"
        with open(output_filename, 'w') as file:
            file.write(processed_text)

        print(f"Success! The {mode}ed text has been saved to: {output_filename}")

    except Exception as e:
        print(f"An unexpected error occurred: {e}")

def main():
    print("--- Professional Caesar Cipher Tool ---")
    
    while True:
        choice = input("\n(E)ncrypt File, (D)ecrypt File, or (Q)uit: ").upper()
        
        if choice == 'Q':
            break
            
        if choice in ['E', 'D']:
            filename = input("Enter the filename (e.g., message.txt): ")
            try:
                shift = int(input("Enter shift value: "))
            except ValueError:
                print("Invalid shift. Please enter a number.")
                continue
            
            mode = 'encrypt' if choice == 'E' else 'decrypt'
            process_file(filename, shift, mode)
        else:
            print("Invalid selection.")

if __name__ == "__main__":
    main()
